```typescript
interface ChatRequest {
  message: string;
  chat_id: string;
  session_id: string;
}

interface ChatResponse {
  message_id: string;
  content: string;
  proposed_actions: ProposedAction[];
  requires_approval: boolean;
}

interface ProposedAction {
  call_id: string;
  tool_name: string;
  friendly_description: string;
  parameters: Record<string, any>;
  estimated_impact: ImpactEstimate;
  warnings?: Warning[];
  dependencies?: string[];
}

interface DispatchRequest {
  selected_actions: SelectedAction[];
  session_id: string;
}

interface SelectedAction {
  call_id: string;
  approved: boolean;
  modified_args?: Record<string, any>;
}

class ApiService {
  private baseURL: string;
  private apiVersion: string = 'v1';

  private getHeaders(): Record<string, string> {
    return {
      'Content-Type': 'application/json',
      'API-Version': this.apiVersion,
      'Authorization': `Bearer ${this.getAuthToken()}`,
      'X-Session-ID': this.getSessionId(),
      'X-Request-ID': this.generateRequestId(),
    };
  }

  async sendMessage(request: ChatRequest): Promise<ChatResponse> {
    const response = await fetch(`${this.baseURL}/api/v1/chat`, {
      method: 'POST',
      headers: this.getHeaders(),
      body: JSON.stringify(request),
    });

    // Check for version deprecation warnings
    const deprecationWarning = response.headers.get('X-API-Deprecation-Warning');
    if (deprecationWarning) {
      console.warn('API Deprecation Warning:', deprecationWarning);
    }

    return response.json();
  }

  async dispatchActions(request: DispatchRequest): Promise<ExecutionResult[]> {
    const response = await fetch(`${this.baseURL}/api/v1/financial/actions/dispatch`, {
      method: 'POST',
      headers: this.getHeaders(),
      body: JSON.stringify(request),
    });
    return response.json();
  }
}
```

```typescript
        </CardDescription>
      </CardHeader>

      <CardContent className="space-y-4">
        {actions.map(action => (
          <ActionCard
            key={action.call_id}

            onSelect={handleSelectAction}
            onEditParams={setEditingParams}

      </CardContent>

      <CardFooter>
        <ActionDispatch

function ActionCard({ action, selected, onSelect }: ActionCardProps) {
```
