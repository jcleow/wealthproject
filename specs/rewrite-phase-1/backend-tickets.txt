```go
type VersionMiddleware struct {
    supportedVersions map[string]bool
    deprecatedVersions map[string]string // version -> deprecation notice
}

func (v *VersionMiddleware) ValidateVersion(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        version := extractVersion(r) // from URL path or header

        if !v.supportedVersions[version] {
            writeError(w, 400, "unsupported_api_version", "API version not supported")
            return
        }

        // Add version to response headers
        w.Header().Set("API-Version", version)

        // Add deprecation warning if applicable
        if warning, deprecated := v.deprecatedVersions[version]; deprecated {
            w.Header().Set("X-API-Deprecation-Warning", warning)
        }

        next.ServeHTTP(w, r)
    })
}

// Route setup with versioning
router.PathPrefix("/api/v1/").Handler(v1Router)
// Future: router.PathPrefix("/api/v2/").Handler(v2Router)
```

**Database Schema:**
```sql
CREATE TABLE chat_sessions (
    session_id UUID PRIMARY KEY,
    user_id UUID NOT NULL,
    state JSONB NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE conversation_history (
    id UUID PRIMARY KEY,
    session_id UUID REFERENCES chat_sessions(session_id),
    role TEXT NOT NULL,
    content TEXT NOT NULL,
    tool_calls JSONB,
    created_at TIMESTAMP DEFAULT NOW()
);
```
